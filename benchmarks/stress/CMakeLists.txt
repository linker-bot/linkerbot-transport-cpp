cmake_minimum_required(VERSION 3.16)
project(local_transport_stress LANGUAGES C CXX)
set(TRANSPORT_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../.." CACHE PATH "Local transport source")
get_filename_component(TRANSPORT_SOURCE "${TRANSPORT_SOURCE}" REALPATH)
if(NOT EXISTS "${TRANSPORT_SOURCE}/include/linkerbot/transport/session.hpp")
    message(FATAL_ERROR "TRANSPORT_SOURCE must be a local communication library source tree")
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT TARGET linkerbot::can)
    set(LINKERBOT_TRANSPORT_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(LINKERBOT_TRANSPORT_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(LINKERBOT_TRANSPORT_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
    set(LINKERBOT_ALLOW_DEPENDENCY_DOWNLOADS OFF CACHE BOOL "" FORCE)
    set(LINKERBOT_TRANSPORT_ENABLE_COVERAGE OFF CACHE BOOL "" FORCE)
    set(LINKERBOT_TRANSPORT_ENABLE_SERIAL OFF CACHE BOOL "" FORCE)
    set(LINKERBOT_TRANSPORT_ENABLE_MODBUS OFF CACHE BOOL "" FORCE)
    add_subdirectory("${TRANSPORT_SOURCE}" local-library)
endif()
foreach(worker throughput profile)
    add_executable(${worker}_worker src/${worker}.cpp)
    target_compile_features(${worker}_worker PRIVATE cxx_std_17)
    target_link_libraries(${worker}_worker PRIVATE linkerbot::can)
    target_link_options(${worker}_worker PRIVATE "-Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/${worker}_worker.map")
endforeach()
add_library(socket_probe SHARED src/socket_probe.c)
target_link_libraries(socket_probe PRIVATE pthread)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/linked-library.txt"
    CONTENT "$<TARGET_FILE:linkerbot-can>\n$<TARGET_PROPERTY:linkerbot-can,LINKERBOT_TRANSPORT_VERSION>\n${TRANSPORT_SOURCE}\n")
